#include <WiFi.h>
#include <PubSubClient.h>
#include "DHT.h"

// --- Network & Broker Settings ---
const char* ssid        = "HiranA07";    
const char* password    = "navi1234";   
const char* mqtt_server = "10.252.47.49";        
const int   mqtt_port   = 1883;

// --- Sensor Pin Definitions ---
#define DHTPIN 4
#define DHTTYPE DHT11

#define S0_PIN 21
#define S1_PIN 22
#define S2_PIN 18
#define S3_PIN 5
#define OUT_PIN 19

#define TRIG_PIN 13
#define ECHO_PIN 14

// TCS3200 Calibration Parameters
int redMin = 30,  redMax = 200;
int greenMin = 30, greenMax = 220;
int blueMin = 25,  blueMax = 180;

DHT dht(DHTPIN, DHTTYPE);
WiFiClient espClient;
PubSubClient client(espClient);

void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi connected!");
  Serial.print("ESP32 IP Address: ");
  Serial.println(WiFi.localIP());
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection...");
    String clientId = "ESP32FabricClient-";
    clientId += String(random(0xffff), HEX);
    
    if (client.connect(clientId.c_str())) {
      Serial.println("connected to Mosquitto!");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" retrying in 5 seconds...");
      delay(5000);
    }
  }
}

float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return -1;
  return (duration * 0.0343) / 2.0;
}

// Advanced Material Classifier (RGB Brightness + Surface Reflection + Humidity Multi-Sensor)
String classifyFabricMaterial(int r, int g, int b, float humidity) {
  int totalBrightness = r + g + b;

  // 1. Polyester: High specular reflection / sheen (high overall RGB intensity)
  if (totalBrightness > 500) {
    return "Polyester";
  } 
  // 2. Silk: High luster with dominant color ratio at close range (1-5 cm)
  else if (totalBrightness > 300 && totalBrightness <= 500) {
    return "Silk";
  } 
  // 3. Cotton: Matte surface, lower light reflection, moisture-retentive properties
  else if (totalBrightness > 50 && totalBrightness <= 300) {
    if (humidity > 65.0) {
      return "Cotton (High Moisture)";
    }
    return "Cotton";
  } 
  else {
    return "Unknown Material";
  }
}

void setup() {
  Serial.begin(115200);

  dht.begin();

  pinMode(S0_PIN, OUTPUT);
  pinMode(S1_PIN, OUTPUT);
  pinMode(S2_PIN, OUTPUT);
  pinMode(S3_PIN, OUTPUT);
  pinMode(OUT_PIN, INPUT);
  digitalWrite(S0_PIN, HIGH);
  digitalWrite(S1_PIN, LOW);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  float distance = getDistance();

  // Trigger Logic: Process fabric within 30 cm
  if (distance > 0 && distance <= 30.0) {
    delay(200);

    // Read TCS3200 RGB
    digitalWrite(S2_PIN, LOW); digitalWrite(S3_PIN, LOW);
    int redFreq = pulseIn(OUT_PIN, LOW, 100000);
    int R = constrain(map(redFreq, redMin, redMax, 255, 0), 0, 255);

    digitalWrite(S2_PIN, HIGH); digitalWrite(S3_PIN, HIGH);
    int greenFreq = pulseIn(OUT_PIN, LOW, 100000);
    int G = constrain(map(greenFreq, greenMin, greenMax, 255, 0), 0, 255);

    digitalWrite(S2_PIN, LOW); digitalWrite(S3_PIN, HIGH);
    int blueFreq = pulseIn(OUT_PIN, LOW, 100000);
    int B = constrain(map(blueFreq, blueMin, blueMax, 255, 0), 0, 255);

    // Read DHT11
    float humidity = dht.readHumidity();
    float temperature = dht.readTemperature();
    if (isnan(humidity)) humidity = 0.0;
    if (isnan(temperature)) temperature = 0.0;

    // Classify Material Type
    String fabricMaterial = classifyFabricMaterial(R, G, B, humidity);

    // Construct Payload
    String payload = "{";
    payload += "\"fabric_present\":true,";
    payload += "\"distance\":" + String(distance, 1) + ",";
    payload += "\"fabric_type\":\"" + fabricMaterial + "\",";
    payload += "\"r\":" + String(R) + ",";
    payload += "\"g\":" + String(G) + ",";
    payload += "\"b\":" + String(B) + ",";
    payload += "\"temperature\":" + String(temperature, 1) + ",";
    payload += "\"humidity\":" + String(humidity, 1);
    payload += "}";

    client.publish("fabric/sensors", payload.c_str());
    Serial.print("Fabric Scanned & Published: ");
    Serial.println(payload);

  } else {
    Serial.print("Idle (> 30cm). Current Distance: ");
    Serial.print(distance);
    Serial.println(" cm");
  }

  delay(1000);
}
